let selectedKey="A",selectedMood="",selectedEmoji="";

function show(id){
 document.querySelectorAll(".screen").forEach(x=>x.classList.remove("active"));
 document.getElementById(id).classList.add("active");
 window.scrollTo(0,0);
}
function unlock(){
 if(document.getElementById("secret").value.trim()==="KAKUREGA02"){
  show("dashboard");
 }else{
  document.getElementById("gateMsg").textContent="Wrong secret code. Try again ♡";
 }
}
function changeMode(){
 selectedKey=document.getElementById("mode").value;
 renderCharacters();
}
function renderCharacters(){
 const container=document.getElementById("characters");
 container.innerHTML="";
 characters.forEach(c=>{
  const card=document.createElement("div");
  card.className="character"+(selections[selectedKey].name===c[0]?" selected":"");
  card.innerHTML=`<span>${c[1]}</span><strong>${c[0]}</strong>`;
  card.onclick=()=>{
   selections[selectedKey].name=c[0];
   selections[selectedKey].bg=c[2];
   selections[selectedKey].accent=c[3];
   renderCharacters();
   document.getElementById("selectedInfo").textContent="Selected: "+c[0];
  };
  container.appendChild(card);
 });
}
function setTheme(bg,accent){
 selections[selectedKey].bg=bg;
 selections[selectedKey].accent=accent;
 document.documentElement.style.setProperty("--panel",bg);
 document.documentElement.style.setProperty("--accent",accent);
}
function applySettings(){
 const s=selections[selectedKey];
 document.documentElement.style.setProperty("--panel",s.bg);
 document.documentElement.style.setProperty("--accent",s.accent);

 "Applied: "+s.name+" ♡";
}
function sendMessage(){

 const name=document.getElementById("chatName").value.trim()||"Me";

 document.getElementById("messages").appendChild(el);
 input.value="";
}
function saveLetter(){

 const p=document.createElement("p");p.textContent=body;

 document.getElementById("letterList").prepend(el);
 document.getElementById("letterText").value="";
}
